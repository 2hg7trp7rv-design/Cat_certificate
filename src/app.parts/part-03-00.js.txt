  renderGame() {
    const pet = this.state.pet
    const now = this.now()
    const virtualDate = new Date(now)
    const routineSleeping = isSleepTime(this.state.routine, virtualDate)
    const isSleeping = this.activity === 'sleeping' || (routineSleeping && !['eating', 'playing', 'petted'].includes(this.activity))
    const scenePhase = isSleeping ? 'night' : this.phase
    const greeting = getGreeting(this.state, this.phase, now)
    const date = new Intl.DateTimeFormat('ja-JP', { month: 'long', day: 'numeric', weekday: 'short' }).format(virtualDate)
    const clock = new Intl.DateTimeFormat('ja-JP', { hour: '2-digit', minute: '2-digit' }).format(virtualDate)
    const bondLevel = Math.max(1, Math.floor(pet.bond / 10) + 1)
    const mealStatus = getMealStatus(this.state, now)
    const growth = getGrowthStage(pet, now)
    const mood = getPetMood(pet, mealStatus, isSleeping)

    this.root.innerHTML = `<main class="game-shell game-shell--${this.phase} ${isSleeping ? 'is-sleeping' : ''}">
      <header class="status-header">
        <div class="status-header__identity"><span class="status-header__date">${date} · ${clock} · ${phaseLabels[this.phase]}</span><div class="status-header__name-row"><h1>${escapeHTML(pet.name)}</h1><span class="growth-chip">${growth.label}</span><span class="bond-chip">${icon('sparkles', 13)} なかよし Lv.${bondLevel}</span></div></div>
        <div class="status-header__actions"><button id="open-journal" class="icon-button" aria-label="思い出を見る">${icon('book', 21)}</button><button id="open-settings" class="icon-button" aria-label="設定を開く">${icon('settings', 21)}</button></div>
      </header>
      <section class="game-content">
        <div class="greeting-card ${mealStatus.overdue ? 'is-alert' : ''}"><span class="greeting-card__dot"></span><p>${escapeHTML(greeting)}</p>${icon(mealStatus.overdue ? 'bell' : 'heart', 16)}</div>
        ${this.roomSceneHTML(pet, scenePhase, mood, growth, isSleeping)}
        ${this.wellbeingHTML(pet, mealStatus, now)}
      </section>
      ${this.coachmarkVisible && this.state.sessionCount <= 2 && !this.playMode && this.activity === 'idle' ? `<div class="coachmark">${icon('sparkles', 17)}<span><strong>まずは触れてみよう</strong><small>${escapeHTML(pet.name)}の顔や背中を指で撫でてみてください。</small></span><button id="close-coach" aria-label="閉じる">${icon('close', 15)}</button></div>` : ''}
      ${this.dockHTML()}
      ${this.foodOpen ? this.foodSheetHTML(pet) : ''}
      ${this.panel === 'journal' ? this.journalHTML(pet) : ''}
      ${this.panel === 'room' ? this.livingHTML() : ''}
      ${this.panel === 'settings' ? this.settingsHTML() : ''}
    </main>`

    this.bindGame()
  }
