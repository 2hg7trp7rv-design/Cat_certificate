      dinnerTime: DEFAULT_ROUTINE.dinnerTime,
      bedtime: DEFAULT_ROUTINE.bedtime,
    }

    this.save()
    this.render()

    window.setTimeout(() => {
      if (this.activity === 'greeting') {
        this.activity = 'idle'
        this.render()
      }
    }, 1800)

    window.setInterval(() => {
      if (!this.state.pet) return
      const now = this.now()
      const next = getTimePhase(new Date(now))
      this.state = applyElapsedTime(this.state, now)
      const needsRender = next !== this.phase
      this.phase = next
      this.save()
      this.maybeShowCareNotice()
      if (needsRender) this.render()
    }, 60_000)

    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState !== 'visible' || !this.state.pet) return
      const now = this.now()
      this.state = applyElapsedTime(this.state, now)
      this.phase = getTimePhase(new Date(now))
      this.save()
      this.render()
      this.maybeShowCareNotice()
    })
  }

  now() {
    return getVirtualNow(this.state)
  }

  virtualDate() {
    return new Date(this.now())
  }

  save() {
    safeStorage.set(STORAGE_KEY, serializeState(this.state))
  }

  announce(message) {
    const live = $('#live-region')
    if (live) live.textContent = message
  }

  render() {
    document.documentElement.style.colorScheme = 'light'
    if (!this.state.onboarded || !this.state.pet) this.renderOnboarding()
    else this.renderGame()
  }
