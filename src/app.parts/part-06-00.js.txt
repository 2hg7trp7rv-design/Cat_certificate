      if (!this.playMode && strokeDistance > 12) this.handlePet()
      lastPoint = null
      strokeDistance = 0
    }
    scene.addEventListener('pointerup', end)
    scene.addEventListener('pointercancel', end)
    scene.addEventListener('pointerleave', () => {
      if (!pointerDown) updateEyeTarget(petStage, this.state.pet.kind, 0, 0)
    })
  }

  spawnTouchHeart(scene, x, y) {
    const heart = document.createElement('span')
    heart.className = 'touch-heart'
    heart.textContent = '♥'
    heart.style.left = `${x}px`
    heart.style.top = `${y}px`
    scene.appendChild(heart)
    window.setTimeout(() => heart.remove(), 920)
  }

  handlePet() {
    if (!this.state.pet || this.playMode || this.activity === 'eating') return
    const result = petCompanion(this.state, this.now())
    this.state = result.state
    this.save()
    this.coachmarkVisible = false
    this.purr(this.state.soundEnabled)
    this.haptic('light')
    this.setTemporaryActivity('petted', 1350)
    this.showToast(result.message, result.bondDelta > 0 ? `なかよし +${result.bondDelta}` : '')
  }

  handleFeed(food) {
    const result = feedCompanion(this.state, food, this.now())
    this.state = result.state
    this.foodOpen = false
    this.playMode = false
    this.save()
    this.softChime(this.state.soundEnabled)
    this.haptic('success')
    this.setTemporaryActivity('eating', 2300)
    this.showToast(result.message, `なかよし +${result.bondDelta}`)
  }

  handlePlayToggle() {
    const pet = this.state.pet
    if (this.playMode) {
      this.playMode = false
      this.activity = 'idle'
      this.render()
      return
    }
    if (pet.energy < 18) {
      this.showToast(`${pet.name}は少し眠たそう。先に休ませてあげよう。`)
      return
    }
    this.foodOpen = false
    this.coachmarkVisible = false
    this.activity = 'playing'
    this.playMode = true
    this.haptic('light')
    this.render()
  }

  handlePlayComplete() {
    if (!this.playMode) return
    const result = playWithCompanion(this.state, this.now())
    this.state = result.state
    this.playMode = false
    this.save()
    this.softChime(this.state.soundEnabled)
    this.haptic('success')
    this.setTemporaryActivity('petted', 1800)
    this.showToast(result.message, `なかよし +${result.bondDelta}`)
  }

  handleRest() {
    this.playMode = false
    this.foodOpen = false
    if (this.activity === 'sleeping') {
      this.activity = 'greeting'
      this.render()
      this.showToast(`${this.state.pet.name}が、ゆっくり目を開けた。`)
      window.setTimeout(() => { this.activity = 'idle'; this.render() }, 1100)
      return
    }
    const result = restCompanion(this.state, this.now())
    this.state = result.state
    this.activity = 'sleeping'
