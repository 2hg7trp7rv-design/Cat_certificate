const safeStorage = {
  get(key) {
    try { return window.localStorage?.getItem(key) ?? null } catch { return null }
  },
  set(key, value) {
    try { window.localStorage?.setItem(key, value); return true } catch { return false }
  },
  remove(key) {
    try { window.localStorage?.removeItem(key); return true } catch { return false }
  },
}

class TailRoomApp {
  constructor(root) {
    this.root = root
    const params = new URLSearchParams(location.search)
    let restored = parseState(safeStorage.get(STORAGE_KEY))
    const demoKind = params.get('demo')
    if (demoKind === 'cat' || demoKind === 'dog') {
      const actualNow = Date.now()
      const demoDays = Math.max(0, Math.min(365, Number(params.get('days') || 0)))
      restored = beginGame(
        restored,
        {
          kind: demoKind,
          name: demoKind === 'dog' ? 'ルカ' : 'こむぎ',
          coat: demoKind === 'dog' ? 'ginger' : 'cream',
          personality: demoKind === 'dog' ? 'curious' : 'gentle',
        },
        '',
        DEFAULT_ROUTINE,
        actualNow,
      )
      restored.pet.createdAt = actualNow - demoDays * DAY
      restored.pet.lastFedAt = actualNow
      restored.lastSeenAt = actualNow
    }
    this.state = applyElapsedTime(restored, getVirtualNow(restored))
    this.phase = getTimePhase(new Date(getVirtualNow(this.state)))
    this.activity = 'greeting'
    this.playMode = false
    this.foodOpen = false
    this.panel = ['settings', 'journal', 'room'].includes(params.get('panel')) ? params.get('panel') : 'home'
    this.coachmarkVisible = true
    this.toastTimer = null
    this.activityTimer = null
    this.playCompleted = false
    this.audioContext = null
    this.lastCareNoticeKey = ''
    this.onboarding = {
      step: 0,
      kind: 'cat',
      coat: 'cream',
      personality: 'gentle',
      name: 'こむぎ',
      playerName: '',
      wakeTime: DEFAULT_ROUTINE.wakeTime,
      breakfastTime: DEFAULT_ROUTINE.breakfastTime,
