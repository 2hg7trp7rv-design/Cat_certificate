      this.save()
      this.render()
      if (permission === 'granted') {
        new Notification(`${this.state.pet.name}の暮らし`, {
          body: '空腹通知の確認ができるようになりました。',
          icon: '/favicon.svg',
        })
        this.showToast('ブラウザ通知を許可しました。')
      } else {
        this.showToast('通知は許可されていません。ゲーム本体はそのまま遊べます。')
      }
    } catch {
      this.showToast('通知の確認に失敗しました。正式版ではiOS通知として実装します。')
    }
  }

  handleAdvanceTime(milliseconds) {
    if (!Number.isFinite(milliseconds) || milliseconds <= 0) return
    this.state = advanceVirtualTime(this.state, milliseconds)
    const now = this.now()
    this.state = applyElapsedTime(this.state, now)
    this.phase = getTimePhase(new Date(now))
    this.activity = 'idle'
    this.playMode = false
    this.foodOpen = false
    this.save()
    this.render()
    const days = Math.floor(milliseconds / DAY)
    const hours = Math.floor(milliseconds / HOUR)
    this.showToast(days >= 1 ? `${days}日後の暮らしへ進めました。` : `${hours}時間後の暮らしへ進めました。`)
    this.maybeShowCareNotice()
  }

  maybeShowCareNotice(force = false) {
    if (!this.state.pet) return
    const now = this.now()
    const status = getMealStatus(this.state, now)
    if (!status.overdue && !force) return
    const dateKey = new Date(status.dueMeal?.at || now).toISOString().slice(0, 10)
    const noticeKey = `${status.dueMeal?.id || 'preview'}-${dateKey}`
    if (!force && noticeKey === this.lastCareNoticeKey) return
    this.lastCareNoticeKey = noticeKey
    const title = status.overdue ? `${this.state.pet.name}の${status.dueMeal.label}` : `${this.state.pet.name}の空腹通知プレビュー`
    const body = status.overdue
      ? `${this.state.pet.name}がお皿を何度も確認しています。予定から${status.minutesLate}分ほど経っています。`
      : `朝または夜のごはんが30分以上遅れると、このように知らせます。`
    if (this.state.notificationEnabled && typeof Notification !== 'undefined' && Notification.permission === 'granted') {
      try { new Notification(title, { body, icon: '/favicon.svg' }) } catch { /* preview enhancement only */ }
    }
    this.showToast(body, status.overdue ? '空腹' : '通知見本')
  }
